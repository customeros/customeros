query GetTimelineEvents($ids: [ID!]!) {
  timelineEvents(ids: $ids) {
    __typename
    ... on PageView {
      id
      application
      startedAt
      endedAt
      engagedTime
      pageUrl
      pageTitle
      orderInSession
      sessionId
      source
      appSource
    }
    ... on Issue {
      id
      subject
      status
      priority
      description
      source
      appSource
    }
    ... on Note {
      id
      content
      createdAt
    }
    ... on InteractionEvent {
      id
      content
      contentType
      channel
      createdAt
    }
  }
}

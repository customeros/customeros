query GetContactById($contactId: ID!, $from: Time!, $size: Int!){
  contact(id: $contactId) {
    id
    timelineEvents(from:$from, size:$size) {
            __typename
            ... on InteractionEvent {
                id
                createdAt
                content
                contentType
                channel
                source
                appSource
                interactionSession {
                  id
                  createdAt
                  updatedAt
                  name
                  type
                  channel
                  status
                  source
                  appSource
                }
                actionItems {
                    id
                    content
                }
            }
          }
  }
}

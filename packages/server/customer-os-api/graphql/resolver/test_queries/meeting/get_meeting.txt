query GetMeeting($meetingId: ID!){
  meeting(id: $meetingId) {
       id
       name
       appSource
       startedAt
       endedAt
       source
       events {
         id
         createdAt
         eventIdentifier
         content
         contentType
         channel
         source
         appSource
         repliesTo {
           id
           eventIdentifier
           content
           contentType
           channel
         }
       }
      status
  }
}

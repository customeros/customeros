query GetIssue($issueId: ID!){
  issue(id: $issueId) {
       id
       createdAt
       updatedAt
       subject
       status
       priority
       description
       tags {
            metadata {
                id
            }
            name
       }
       interactionEvents {
            id
       }
       comments {
            id
            content
            contentType
            createdBy {
                id
            }
            externalLinks {
                externalId
            }
       }
       externalLinks {
            type
            syncDate
            externalId
            externalUrl
            externalSource
       }
  }
}
